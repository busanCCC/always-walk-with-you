```typescript
function JournalDetailScreen({ route }: Props) {
  const { journalId } = route.params;
  const { data: journal, isLoading } = useQuery(
    ['journal', journalId],
    () => fetchJournal(journalId)
  );
  
  if (isLoading) return <LoadingIndicator />;
  
  return (
    <ScrollView>
      <JournalHeader
        date={journal.date}
        emotion={journal.emotion}
      />
      <JournalContent
        mode={journal.mode}
        content={journal.content}
      />
      <CommentsSection journalId={journalId} />
    </ScrollView>
  );
}
```
