function GroupDetailScreen({ route }: Props) {
  const { groupId } = route.params;
  const [activeTab, setActiveTab] = useState('journals');
  
  return (
    <View>
      <GroupHeader groupId={groupId} />
      <TabBar
        tabs={['일기', '멤버']}
        activeTab={activeTab}
        onTabPress={setActiveTab}
      />
      {activeTab === 'journals' && <SharedJournalsList groupId={groupId} />}
      {activeTab === 'members' && <MembersList groupId={groupId} />}
    </View>
  );
}
