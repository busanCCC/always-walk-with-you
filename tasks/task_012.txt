```typescript
function useChapelBulletin() {
  return useQuery(['chapelBulletin'], async () => {
    const { data, error } = await supabase
      .from('chapel_bulletins')
      .select('*')
      .order('date', { ascending: false })
      .limit(1);
      
    if (error) throw error;
    return data[0];
  });
}
```
3. 홈 화면에 주보 카드 컴포넌트 구현:
```typescript
function ChapelBulletinCard() {
  const { data, isLoading } = useChapelBulletin();
  
  if (isLoading) return <LoadingIndicator />;
  if (!data) return null;
  
  return (
    <Card>
      <Text className="font-bold text-lg">{data.title}</Text>
      <Text className="text-gray-600">{formatDate(data.date)}</Text>
      <Text className="mt-2">{data.scripture}</Text>
      <Button
        title="자세히 보기"
        onPress={() => navigateToDetail(data.id)}
      />
    </Card>
  );
}
```
