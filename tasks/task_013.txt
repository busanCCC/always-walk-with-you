```typescript
function CommentsSection({ journalId }: Props) {
  const { data: comments, isLoading } = useQuery(
    ['comments', journalId],
    () => fetchComments(journalId)
  );
  
  const [newComment, setNewComment] = useState('');
  
  const addCommentMutation = useMutation(addComment, {
    onSuccess: () => {
      queryClient.invalidateQueries(['comments', journalId]);
      setNewComment('');
    }
  });
  
  return (
    <View>
      <FlatList
        data={comments}
        renderItem={({ item }) => <CommentItem comment={item} />}
      />
      <View className="flex-row">
        <Input
          value={newComment}
          onChangeText={setNewComment}
          placeholder="댓글을 입력하세요"
          className="flex-1"
        />
        <Button
          title="등록"
          onPress={() => addCommentMutation.mutate({
            journalId,
            content: newComment
          })}
          loading={addCommentMutation.isLoading}
        />
      </View>
    </View>
  );
}
```

```typescript
function LikeButton({ journalId }: Props) {
  const { data: isLiked } = useQuery(
    ['isLiked', journalId],
    () => checkIfLiked(journalId)
  );
  
  const likeMutation = useMutation(toggleLike, {
    onSuccess: () => {
      queryClient.invalidateQueries(['isLiked', journalId]);
    }
  });
  
  return (
    <TouchableOpacity
      onPress={() => likeMutation.mutate(journalId)}
      className={`p-2 ${isLiked ? 'bg-red-100' : 'bg-gray-100'}`}
    >
      <Text className={isLiked ? 'text-red-500' : 'text-gray-500'}>
        ❤️ {isLiked ? '좋아요 취소' : '좋아요'}
      </Text>
    </TouchableOpacity>
  );
}
```
