```typescript
function ErrorBoundary({ children }: Props) {
  const [error, setError] = useState<Error | null>(null);
  
  if (error) {
    return (
      <ErrorScreen
        error={error}
        onReset={() => setError(null)}
      />
    );
  }
  
  return (
    <ErrorHandler onError={setError}>
      {children}
    </ErrorHandler>
  );
}
```
