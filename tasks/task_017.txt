```sql
-- 일기 테이블 RLS 예시
CREATE POLICY "사용자는 자신의 일기만 볼 수 있음"
  ON journals
  FOR SELECT
  USING (auth.uid() = user_id OR
         EXISTS (
           SELECT 1 FROM groups g
           JOIN group_members gm ON g.id = gm.group_id
           WHERE gm.user_id = auth.uid()
           AND g.id = ANY(journals.shared_groups)
         ));
```
